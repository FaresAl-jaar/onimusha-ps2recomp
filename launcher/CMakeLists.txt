cmake_minimum_required(VERSION 3.24)
project(OnimushaLauncher VERSION 1.0 LANGUAGES C CXX RC)
if(NOT MSVC OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
  message(FATAL_ERROR "Select a 64-bit MSVC toolchain.")
endif()
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
if(CMAKE_VERSION VERSION_GREATER_EQUAL 4.0)
  set(CMAKE_POLICY_VERSION_MINIMUM 3.10)
endif()
include(FetchContent)
# An archive in vendor/ is preferred so repeat builds also work offline.
function(launcher_dependency name archive sha remote)
  if(EXISTS "${PROJECT_SOURCE_DIR}/vendor/${archive}")
    set(source "${PROJECT_SOURCE_DIR}/vendor/${archive}")
  else()
    set(source "${remote}")
  endif()
  FetchContent_Declare(${name} URL "${source}" URL_HASH "SHA256=${sha}" DOWNLOAD_EXTRACT_TIMESTAMP TRUE)
endfunction()
set(SDL_SHARED OFF CACHE BOOL "" FORCE)
set(SDL_STATIC ON CACHE BOOL "" FORCE)
set(SDL_TEST OFF CACHE BOOL "" FORCE)
set(SDL_TESTS OFF CACHE BOOL "" FORCE)
set(SDL2_DISABLE_INSTALL ON CACHE BOOL "" FORCE)
set(SDL_FORCE_STATIC_VCRT ON CACHE BOOL "" FORCE)
set(JSON_BuildTests OFF CACHE BOOL "" FORCE)
launcher_dependency(sdl2 SDL2-2.32.10.tar.gz 03f9d7c191a837525c9cda6406af2f2e48be02b5e7eb03d949cc9f1e9ca41c8b https://codeload.github.com/libsdl-org/SDL/tar.gz/refs/tags/release-2.32.10)
launcher_dependency(imgui imgui-1.91.9b.tar.gz 8e1bbc76c71d74fef2fb85db7e7ca8eba13d6a86623c54992b60162db554ffdb https://codeload.github.com/ocornut/imgui/tar.gz/refs/tags/v1.91.9b)
launcher_dependency(json json-3.11.3.tar.gz 0d8ef5af7f9794e3263480193c491549b2ba6cc74bb018906202ada498a79406 https://codeload.github.com/nlohmann/json/tar.gz/refs/tags/v3.11.3)
FetchContent_MakeAvailable(sdl2 imgui json)
add_library(launcher_imgui STATIC
  ${imgui_SOURCE_DIR}/imgui.cpp ${imgui_SOURCE_DIR}/imgui_draw.cpp
  ${imgui_SOURCE_DIR}/imgui_widgets.cpp ${imgui_SOURCE_DIR}/imgui_tables.cpp
  ${imgui_SOURCE_DIR}/misc/cpp/imgui_stdlib.cpp
  ${imgui_SOURCE_DIR}/backends/imgui_impl_sdl2.cpp
  ${imgui_SOURCE_DIR}/backends/imgui_impl_sdlrenderer2.cpp)
target_include_directories(launcher_imgui PUBLIC ${imgui_SOURCE_DIR} ${imgui_SOURCE_DIR}/backends)
target_link_libraries(launcher_imgui PUBLIC SDL2::SDL2-static)
add_library(launcher_core STATIC bindings.cpp settings.cpp launch.cpp iso.cpp)
target_compile_definitions(launcher_core PUBLIC SDL_MAIN_HANDLED UNICODE _UNICODE NOMINMAX WIN32_LEAN_AND_MEAN)
target_link_libraries(launcher_core PUBLIC SDL2::SDL2-static nlohmann_json::nlohmann_json ole32 shell32 uuid)
add_executable(OnimushaLauncher WIN32 main.cpp ui_shell.cpp ui_game.cpp ui_options.cpp ui_controls.cpp launcher.rc)
target_link_libraries(OnimushaLauncher PRIVATE launcher_core launcher_imgui)
target_link_options(OnimushaLauncher PRIVATE "/MANIFEST:EMBED" "/MANIFESTINPUT:${CMAKE_CURRENT_SOURCE_DIR}/launcher.manifest")
set_target_properties(OnimushaLauncher PROPERTIES RUNTIME_OUTPUT_DIRECTORY_RELEASE "${CMAKE_BINARY_DIR}/OnimushaPC")
foreach(target launcher_core OnimushaLauncher)
  target_compile_options(${target} PRIVATE $<$<COMPILE_LANGUAGE:CXX>:/W4> $<$<COMPILE_LANGUAGE:CXX>:/utf-8> $<$<COMPILE_LANGUAGE:CXX>:/permissive->)
endforeach()
include(CTest)
if(BUILD_TESTING)
  add_executable(launcher_tests tests.cpp)
  target_compile_options(launcher_tests PRIVATE /W4 /utf-8)
  target_link_libraries(launcher_tests PRIVATE launcher_core)
  add_test(NAME launcher_contract COMMAND launcher_tests)
endif()
